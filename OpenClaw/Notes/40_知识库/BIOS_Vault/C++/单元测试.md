- 被测函数

```C++
enum class CommandType{
    Help,
    Quit,
    Add,
    List,
    Find,
    unknown
};

struct Command{
    std::string name;
    std::vector<std::string> args;
    std::string raw;
    bool empty() const{
	  return this->name.empty();  
    }
};

class CommandParser{
    //pubilc start
    public :
        Command parseLine(const std::string& inputLine);
    //public end  

    //private start
    private :
        //variable
        bool caseInsensitive=true;

        //function
        std::string trimCopy(const std::string& inputLine);
        std::vector<std::string> splitWhitespace(const std::string& inputLine);
        std::string toLowerCopy(const std::string& inputLine);
    //private end

};
std::string CommandParser::trimCopy(const std::string& inputLine){
    std::string::size_type left=0;
    std::string::size_type right=0;
    std::string::size_type len=0;

    const char* whitespaceSet=" \t\r\n";
    left=inputLine.find_first_not_of(whitespaceSet);
    if(left==std::string::npos){
        return "";
    }
    right=inputLine.find_last_not_of(whitespaceSet);
    len=right-left+1;
    return inputLine.substr(left,len);

}
```
- 测试用例
`BookLibrary\src\tests\CommandParserTest.cpp`

```C++
#define DOCTEST_CONFIG_IMPLEMENT_WITH_MAIN
#include "third_party/doctest/doctest.h"

// 测试专用：把 private 暴露出来（只用于单元测试）

#define private public
#include "src/cli/CommandParser.h"
#undef private

TEST_CASE("trimCopy: empty input -> empty") {
    CommandParser p;
    CHECK(p.trimCopy("") == "");
}

TEST_CASE("trimCopy: all whitespace -> empty") {
    CommandParser p;
    CHECK(p.trimCopy("   ") == "");
    CHECK(p.trimCopy("\t\r\n") == "");
    CHECK(p.trimCopy(" \t \r\n ") == "");
}

TEST_CASE("trimCopy: no trim needed") {
    CommandParser p;
    CHECK(p.trimCopy("abc") == "abc");
    CHECK(p.trimCopy("a b c") == "a b c");
}

TEST_CASE("trimCopy: trim left/right/both") {
    CommandParser p;
    CHECK(p.trimCopy("   abc") == "abc");
    CHECK(p.trimCopy("abc   ") == "abc");
    CHECK(p.trimCopy("  abc  ") == "abc");
    CHECK(p.trimCopy("\t abc \r\n") == "abc");
    CHECK(p.trimCopy("  a b  ") == "a b");
}
```
